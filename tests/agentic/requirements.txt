# Client libraries the agentic lane drives in-process, layered over the project
# environment by `uv run --with-requirements` (see tests/agentic/README.md).
#
# Kept out of pyproject.toml: uv.lock's single universal resolution would pin these
# to whatever also satisfies the gateway. Only tests/agentic/ may import them; other
# modules are skipped when their client is missing (see tests/agentic/conftest.py).
#
# Installing them into .venv instead takes `uv pip install --no-config -r`: without
# it uv applies pyproject.toml's override-dependencies, which drop the `rich` and
# `typer` that agno and livekit-agents import.

# tests/agentic/test_langchain.py
langchain-openai
langchain-anthropic

# tests/agentic/test_pydantic_ai.py
pydantic-ai-slim[openai]

# tests/agentic/test_wyoming_audio.py -- the Home Assistant voice protocol client
# that talks to the wyoming-openai container, not a gateway client itself.
wyoming

# tests/agentic/test_litellm.py
litellm

# tests/agentic/test_openai_agents.py -- the OpenAI Agents SDK, whose RealtimeRunner
# is the lane's only client for /v1/realtime, tool loop included.
openai-agents[realtime]

# tests/agentic/test_livekit.py -- the media framework docs/api_openai_realtime.md
# names first for putting WebRTC or SIP in front of this gateway. The plugin pulls
# livekit-agents and the livekit native runtime in with it.
livekit-plugins-openai

# tests/agentic/test_pipecat.py -- the second framework that page names. The
# `openai` extra is what ships its Realtime service.
pipecat-ai[openai]

# tests/agentic/test_agno.py -- the only framework that provisions a vector store
# for itself, through the same client and base URL it chats with.
agno

# llama-index-llms-openai is deliberately NOT here while every release of it
# caps `openai<3`: openai-agents and pydantic-ai-slim[openai] require `openai>=3`,
# so uv walks it back to 0.1.x, whose llama-index-core 0.10 is built on
# pydantic.v1, which cannot import on Python 3.14 -- and a floor instead walks
# those two clients back. tests/agentic/test_llama_index.py runs under its own
# overlay until the cap lifts, then its line returns here:
#   uv run --with llama-index-llms-openai pytest tests/agentic/test_llama_index.py --agentic

# `ollama` is deliberately NOT here either, and tests/agentic/test_ollama_sdk.py
# imports it all the same: it is already a `test` group dependency in
# pyproject.toml, because tests/test_ollama_*.py drive the same official client
# against the in-process app. Repeating it here would layer a second resolution of
# the same library over the one the gateway's own suite uses, and the two lanes
# would stop agreeing about which client version they proved.

# inspect-ai is deliberately NOT here, and tests/agentic/test_inspect_ai.py drives
# it in a container instead. It depends on aioboto3, whose every release pins
# aiobotocore to a 2.x version, which pins botocore below this project's
# `botocore>=1.43.30` floor -- and the lane's gateway runs on the interpreter this
# overlay is layered onto, so the downgrade breaks routes the gateway serves
# (`POST /v1/vector_stores` answers 500 on a missing `requestMetadata` shape).
# Constraining it is not an option either: `inspect-ai>=0.3.255` and
# `aiobotocore>=3` resolve as unsatisfiable, and dropping that floor makes uv walk
# inspect-ai back to 0.3.69. See the "inspect" image group in _tools.py.
